const BYBIT_MAP = { GOLD: 'PAXGUSDT', BTC: 'BTCUSDT', ETH: 'ETHUSDT', SOL: 'SOLUSDT', XRP: 'XRPUSDT', SUI: 'SUIUSDT' }; async function loadChart(asset) { const sym = BYBIT_MAP[asset] || 'PAXGUSDT'; const baseUrl = 'https://bybit.com'; try { const res = await fetch(baseUrl + sym + '&interval=15&limit=55'); const json = await res.json(); if (json.result && json.result.list) { const candles = json.result.list.reverse().map(c => ({ time: Math.floor(Number(c[0]) / 1000), open: parseFloat(c[1]), high: parseFloat(c[2]), low: parseFloat(c[3]), close: parseFloat(c[4]) })); drawCandles(candles); } } catch (e) { console.error(e); } }
